CREATE TABLE Candidat(
   id_candidat VARCHAR(50) ,
   nom_ct VARCHAR(50)  NOT NULL,
   prenom_ct VARCHAR(50)  NOT NULL,
   date_de_naissance DATE NOT NULL,
   region VARCHAR(50)  NOT NULL,
   ville VARCHAR(50) ,
   commune VARCHAR(50)  NOT NULL,
   adresse_ct VARCHAR(50)  NOT NULL,
   nom_pere VARCHAR(50)  NOT NULL,
   occupation_pere VARCHAR(50)  NOT NULL,
   prenom_pere VARCHAR(100)  NOT NULL,
   tel_pere VARCHAR(20)  NOT NULL,
   statut_pere VARCHAR(50)  NOT NULL,
   nom_mere VARCHAR(50)  NOT NULL,
   prenom_mere VARCHAR(50)  NOT NULL,
   occupation_mere VARCHAR(50)  NOT NULL,
   photo_ct VARCHAR(200)  NOT NULL,
   tel_mere VARCHAR(50)  NOT NULL,
   statut_mere VARCHAR(50)  NOT NULL,
   nbr_freres INTEGER NOT NULL,
   nbr_soeurs INTEGER NOT NULL,
   rang INTEGER NOT NULL,
   autres_informations VARCHAR(200)  NOT NULL,
   genre VARCHAR(10)  NOT NULL,
   score_au_bacc VARCHAR(200)  NOT NULL,
   test_entree VARCHAR(200)  NOT NULL,
   photo_de_famille VARCHAR(200)  NOT NULL,
   motivation VARCHAR(200)  NOT NULL,
   visite_a_domicile VARCHAR(200)  NOT NULL,
   orientation VARCHAR(200)  NOT NULL,
   resultat_L0 VARCHAR(200)  NOT NULL,
   resultat_L1 VARCHAR(200)  NOT NULL,
   resultat_L2 VARCHAR(200)  NOT NULL,
   resultat_L3 VARCHAR(200)  NOT NULL,
   educateur VARCHAR(50)  NOT NULL,
   referent_ong VARCHAR(50)  NOT NULL,
   en_cas_urgence VARCHAR(100)  NOT NULL,
   club VARCHAR(100)  NOT NULL,
   foyer VARCHAR(50)  NOT NULL,
   autonomie VARCHAR(200)  NOT NULL,
   universite VARCHAR(100)  NOT NULL,
   parcours VARCHAR(100)  NOT NULL,
   stage_L0 VARCHAR(200)  NOT NULL,
   stage_L1 VARCHAR(200)  NOT NULL,
   stage_L2 VARCHAR(200)  NOT NULL,
   stage_L3 VARCHAR(200)  NOT NULL,
   soutenance VARCHAR(50)  NOT NULL,
   categorie VARCHAR(100)  NOT NULL,
   embauche VARCHAR(200)  NOT NULL,
   statut_ct VARCHAR(50)  NOT NULL,
   mise_a_jour VARCHAR(50)  NOT NULL,
   date_mise_a_jour_ct DATE NOT NULL,
   bourse VARCHAR(100)  NOT NULL,
   batch VARCHAR(50)  NOT NULL,
   quartile VARCHAR(50)  NOT NULL,
   lieu_de_naissance VARCHAR(50) ,
   cni VARCHAR(12)  NOT NULL,
   tel_ct VARCHAR(50)  NOT NULL,
   mail_ct VARCHAR(50)  NOT NULL,
   PRIMARY KEY(id_candidat)
);

CREATE TABLE Note(
   id_note INTEGER,
   date_note DATE NOT NULL,
   titre_note VARCHAR(50)  NOT NULL,
   contenu_note VARCHAR(200)  NOT NULL,
   id_candidat VARCHAR(50)  NOT NULL,
   PRIMARY KEY(id_note),
   FOREIGN KEY(id_candidat) REFERENCES Candidat(id_candidat)
);

CREATE TABLE Utilisateur(
   id_ut INTEGER,
   fonction VARCHAR(100) ,
   nom_ut VARCHAR(100)  NOT NULL,
   prenom_ut VARCHAR(100) ,
   mot_de_passe VARCHAR(100)  NOT NULL,
   nouveau_mot_de_passe VARCHAR(100)  NOT NULL,
   photo_ut VARCHAR(100)  NOT NULL,
   PRIMARY KEY(id_ut)
);

CREATE TABLE Notification(
   id_candidat VARCHAR(50) ,
   id_ut INTEGER,
   historique VARCHAR(50)  NOT NULL,
   date_historique DATE NOT NULL,
   PRIMARY KEY(id_candidat, id_ut),
   FOREIGN KEY(id_candidat) REFERENCES Candidat(id_candidat),
   FOREIGN KEY(id_ut) REFERENCES Utilisateur(id_ut)
);
